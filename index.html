Database Schema (8 tables)

-- 1. USERS
CREATE TABLE users (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100),
    role ENUM('customer', 'admin') DEFAULT 'customer',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_user_email (email)
);

-- 2. CATEGORIES
CREATE TABLE categories (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    slug VARCHAR(100) NOT NULL UNIQUE
);

-- 3. PRODUCTS
CREATE TABLE products (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    category_id INT UNSIGNED,
    name VARCHAR(255) NOT NULL,
    description TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (category_id) REFERENCES categories(id),
    FULLTEXT idx_product_search (name, description)
);

-- 4. PRODUCT VARIANTS
CREATE TABLE product_variants (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    product_id BIGINT UNSIGNED NOT NULL,
    sku VARCHAR(50) NOT NULL UNIQUE,
    attributes JSON,
    price DECIMAL(10, 2) NOT NULL,
    stock_quantity INT NOT NULL DEFAULT 0,
    FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE,
    INDEX idx_variant_price (price)
);

-- 5. COUPONS
CREATE TABLE coupons (
    id INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(50) NOT NULL UNIQUE,
    discount_type ENUM('percent', 'fixed') NOT NULL,
    discount_value DECIMAL(10, 2) NOT NULL,
    min_order_amount DECIMAL(10, 2) DEFAULT 0.00,
    valid_from DATETIME,
    valid_until DATETIME,
    is_active BOOLEAN DEFAULT TRUE,
    INDEX idx_coupon_code (code)
);

-- 6. ORDERS
CREATE TABLE orders (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    coupon_id INT UNSIGNED DEFAULT NULL,
    total_amount DECIMAL(10, 2) NOT NULL,
    status ENUM('pending','paid','shipped','delivered','cancelled') DEFAULT 'pending',
    shipping_address TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (coupon_id) REFERENCES coupons(id),
    INDEX idx_order_date (created_at),
    INDEX idx_order_user (user_id)
);

-- 7. ORDER ITEMS
CREATE TABLE order_items (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    order_id BIGINT UNSIGNED NOT NULL,
    variant_id BIGINT UNSIGNED NOT NULL,
    quantity INT NOT NULL,
    unit_price DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
    FOREIGN KEY (variant_id) REFERENCES product_variants(id)
);

-- 8. REVIEWS
CREATE TABLE reviews (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT UNSIGNED NOT NULL,
    product_id BIGINT UNSIGNED NOT NULL,
    rating TINYINT UNSIGNED CHECK (rating BETWEEN 1 AND 5),
    comment TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id),
    FOREIGN KEY (product_id) REFERENCES products(id),
    UNIQUE KEY unique_user_product_review (user_id, product_id),
    INDEX idx_review_product (product_id, created_at)
);

Top 5 Best-Selling Products (revenue, last 3 months)

SELECT
    p.name AS product_name,
    SUM(oi.quantity * oi.unit_price) AS total_revenue
FROM order_items oi
JOIN orders o ON oi.order_id = o.order_id
JOIN product_variants pv ON oi.variant_id = pv.variant_id
JOIN products p ON pv.product_id = p.product_id
WHERE o.created_at >= NOW() - INTERVAL 3 MONTH
  AND o.status NOT IN ('cancelled', 'pending')
GROUP BY p.product_id, p.name
ORDER BY total_revenue DESC
LIMIT 5;

MySQL: partitioning by date (common interview scenario)

CREATE TABLE orders (
  id BIGINT UNSIGNED,
  user_id BIGINT,
  total DECIMAL(10,2),
  created_at DATETIME NOT NULL,
  PRIMARY KEY (id, created_at)   -- PK MUST include partition key
)
PARTITION BY RANGE (TO_DAYS(created_at)) (
  PARTITION p202601 VALUES LESS THAN (TO_DAYS('2026-02-01')),
  PARTITION p202602 VALUES LESS THAN (TO_DAYS('2026-03-01')),
  PARTITION pmax      VALUES LESS THAN MAXVALUE
);

-- Drop an old month instantly (vs DELETE which scans):
ALTER TABLE orders DROP PARTITION p202601;

-- Pruning in action: only p202602 is read
EXPLAIN SELECT * FROM orders
WHERE created_at >= '2026-02-10' AND created_at < '2026-02-20';

MySQL: backup & point-in-time restore

# 1) Logical backup (consistent, no lock) — small/medium DBs
mysqldump -u root -p --single-transaction \
  --routines --triggers --events myapp > backup.sql

# 2) Restore
mysql -u root -p myapp < backup.sql

# 3) Point-in-time recovery with the binary log
#    (after restoring the backup above)
mysqlbinlog --stop-datetime="2026-07-08 14:00:00" \
  /var/lib/mysql/binlog.000123 | mysql -u root -p

# Big DBs: use Percona XtraBackup (hot physical backup) instead of mysqldump.

MySQL: reading EXPLAIN (real example)

EXPLAIN
SELECT p.name, SUM(oi.quantity * oi.unit_price) AS revenue
FROM orders o
JOIN order_items oi ON oi.order_id = o.id
JOIN product_variants pv ON oi.variant_id = pv.id
JOIN products p ON pv.product_id = p.id
WHERE o.created_at >= '2026-04-08'
  AND o.status = 'paid'
GROUP BY p.id;

-- Look at the 'type' and 'key' columns:
-- type: ALL  -> full scan, NO index used  (bad, add one)
-- type: ref  -> index lookup used         (good)
-- key:  idx_order_date -> the index MySQL picked
-- rows: smaller is better; big number = scan

If type is ALL, add INDEX idx_status_date (status, created_at) so the WHERE filters through the index instead of scanning every row.

Q: products has 1M+ rows. Improve name/category search?

Use Laravel Scout + Typesense. Typesense keeps data in memory with full-text indexing, so search is very fast.

  • Install Typesense server (local or remote).
  • Install Scout + Typesense package in Laravel.
  • Add Typesense credentials in .env.
  • Add Searchable trait to the Product model.
  • Import: php artisan scout:import "App\Models\Product".
  • Search: Product::search('laptop')->get().

Beginner React (≈6 months). Build UIs from components.

Sample Component

import { useState, useEffect } from 'react';

function UserList() {
  const [users, setUsers] = useState([]);

  useEffect(() => {
    fetch('/api/users')
      .then(res => res.json())
      .then(data => setUsers(data));
  }, []);

  return (
    <ul>
      {users.map(u => (
        <li key={u.id}>{u.name}</li>
      ))}
    </ul>
  );
}

export default UserList;
